### 1. Application Source Code

**Code Added:**
```javascript
/**
 * Android Back Button Handler
 * Called by MainActivity.onBackPressed() via JavaScript bridge
 * Returns "handled" if the back action was processed by the web app
 */
window.lifehubHandleBackButton = function() {
  // Close any open modals first
  const openDialogs = Array.from(document.querySelectorAll('dialog[open]'));
  if (openDialogs.length > 0) {
    openDialogs[0].close();
    return 'handled';
  }
  
  // Close mobile navigation if open
  if (document.body.classList.contains('nav-open')) {
    document.body.classList.remove('nav-open');
    return 'handled';
  }
  
  // Navigate back if not on dashboard
  if (state.route !== 'dashboard') {
    navigate('dashboard');
    return 'handled';
  }
  
  // Default: let Android handle it (may exit app)
  return null;
};
```

```java
package com.raizen.lifehub;
import android.os.Build;
import android.os.Bundle;
import com.getcapacitor.BridgeActivity;
import com.getcapacitor.capacitor.App;

public class MainActivity extends BridgeActivity {
  @Override
  protected void onCreate(Bundle savedInstanceState) {
    super.onCreate(savedInstanceState);
  }

  @Override
  public void onBackPressed() {
    // Let the web app handle back navigation first via JavaScript
    this.bridge.eval("window.lifehubHandleBackButton && window.lifehubHandleBackButton()", (value) -> {
      if (value == null || !value.toString().equals("handled")) {
        super.onBackPressed();
      }
    });
  }
}
```
